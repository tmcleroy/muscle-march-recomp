set(WP_GENERATED "${CMAKE_SOURCE_DIR}/build/wiiparty/generated")

file(GLOB WP_DSP_SOURCES CONFIGURE_DEPENDS "${WP_GENERATED}/dsp/*.cpp")
if(WP_DSP_SOURCES)
    target_sources(wp_engine PRIVATE ${WP_DSP_SOURCES})
    set_source_files_properties(${WP_DSP_SOURCES} PROPERTIES COMPILE_OPTIONS -O2)
else()
    target_sources(wp_engine PRIVATE "${CMAKE_SOURCE_DIR}/engine/src/audio/no_dsp_code.cpp")
endif()

file(GLOB WP_DOL_SOURCES CONFIGURE_DEPENDS "${WP_GENERATED}/dol/*.cpp")
file(GLOB_RECURSE WP_MODULE_SOURCES CONFIGURE_DEPENDS "${WP_GENERATED}/modules/*.cpp")
if(NOT WP_DOL_SOURCES)
    message(STATUS "No generated code in ${WP_GENERATED}/dol: run recompiler/recomp.py to build the game")
    return()
endif()

add_library(wiiparty_code STATIC game.cpp ${WP_DOL_SOURCES})
set_source_files_properties(${WP_DOL_SOURCES} ${WP_MODULE_SOURCES} PROPERTIES COMPILE_OPTIONS ${WP_CODE_OPT})
target_include_directories(wiiparty_code PUBLIC "${WP_GENERATED}/dol")
if(WP_MODULE_SOURCES)
    target_sources(wiiparty_code PRIVATE ${WP_MODULE_SOURCES})
else()
    target_sources(wiiparty_code PRIVATE "${CMAKE_SOURCE_DIR}/engine/src/core/no_modules.cpp")
endif()
target_link_libraries(wiiparty_code PUBLIC wp_engine)
target_compile_options(wiiparty_code PRIVATE -Wall -Wextra -Wno-unused-label -Wno-unused-parameter -Wno-clobbered)

add_executable(wiiparty)
if(WIN32)
    target_sources(wiiparty PRIVATE res/wiiparty.rc)
endif()
target_link_libraries(wiiparty PRIVATE wp_app "$<LINK_GROUP:RESCAN,wiiparty_code,wp_engine>")
target_compile_options(wiiparty PRIVATE -Wall -Wextra)
if(WIN32 AND NOT WP_CONSOLE)
    set_target_properties(wiiparty PROPERTIES WIN32_EXECUTABLE ON)
endif()
if(TARGET SDL3::SDL3)
    add_custom_command(TARGET wiiparty POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:wiiparty>)
endif()

add_executable(lifted_tests "${CMAKE_SOURCE_DIR}/tests/wiiparty/lifted_tests.cpp")
target_link_libraries(lifted_tests PRIVATE "$<LINK_GROUP:RESCAN,wiiparty_code,wp_engine>")
add_test(NAME lifted_tests COMMAND lifted_tests)
